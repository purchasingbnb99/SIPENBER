/** * ==================================================================== * SIPENBER - BACKEND GOOGLE APPS SCRIPT (Code.gs) * Disesuaikan khusus untuk Spreadsheet "Database SIPENBER" * - Tab Utama : "DataBerkas" (Fallback ke "Berkas") * - Tab Logbook: "Logbook" * ==================================================================== */ // KATA SANDI AKSES var ADMIN_PASSWORD = "bnbzipper"; var PENERIMA_PASSWORD = "Novita"; // NAMA SHEET DATABASE var PRIMARY_SHEET_NAME = "DataBerkas"; var ALTERNATE_SHEET_NAME = "Berkas"; var SHEET_LOGBOOK = "Logbook"; /** * Menangani permintaan HTTP GET dari Frontend SIPENBER */ function doGet(e) { try { setupSheetsIfNotExist(); var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'getBerkas'; if (action === 'getLogs') { return createJsonResponse(getLogsData()); } return createJsonResponse(getBerkasData()); } catch (err) { return createJsonResponse({ status: 'error', message: err.toString() }); } } /** * Menangani permintaan HTTP POST dari Frontend SIPENBER */ function doPost(e) { try { setupSheetsIfNotExist(); var contents = {}; if (e && e.postData && e.postData.contents) { contents = JSON.parse(e.postData.contents); } var action = contents.action || ''; switch (action) { case 'verifyAdminPassword': var isAdminPassValid = (contents.password === ADMIN_PASSWORD); return createJsonResponse({ status: isAdminPassValid ? 'success' : 'error', success: isAdminPassValid }); case 'verifyPenerimaPassword': var isPenerimaPassValid = (String(contents.password || '').toLowerCase() === PENERIMA_PASSWORD.toLowerCase()); return createJsonResponse({ status: isPenerimaPassValid ? 'success' : 'error', success: isPenerimaPassValid }); case 'createBerkasBatch': return createJsonResponse(saveBerkasBatch(contents.items || [])); case 'updatePenerimaStatus': return createJsonResponse(updatePenerimaStatus(contents.id, contents.isCompleted, contents.notes, contents.noPo)); case 'updatePenerimaStatusBatch': return createJsonResponse(updatePenerimaStatusBatch(contents.updates || [])); case 'updateDocumentDetails': return createJsonResponse(updateDocumentDetails(contents.id, contents.document)); case 'unlockDocument': return createJsonResponse(unlockDocument(contents.id, contents.noPo)); case 'deleteDocument': return createJsonResponse(deleteDocument(contents.id, contents.noPo)); case 'deleteGroupDate': return createJsonResponse(deleteGroupDate(contents.ids || [], contents.pos || [])); case 'addLogbookEntry': return createJsonResponse(addLogbookEntry(contents.logData)); case 'clearLogbook': return createJsonResponse(clearLogbook(contents.idsToDelete || [])); default: return createJsonResponse({ status: 'error', message: 'Aksi POST tidak dikenali: ' + action }); } } catch (err) { return createJsonResponse({ status: 'error', message: err.toString() }); } } /** * Mengambil seluruh daftar berkas dari sheet DataBerkas * Otomatis melengkapi ID unik berbasis baris di Kolom A jika sebelumnya kosong. */ function getBerkasData() { var sheet = getBerkasSheet(); var data = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < data.length; i++) { var row = data[i]; if (row[1] || row[2] || row[3]) { var currentId = String(row[0] || '').trim(); var noPoVal = String(row[2] || '').trim(); // Fix empty or duplicate ID by injecting line-item ID if (!currentId || currentId === 'undefined' || currentId === 'null') { var cleanPo = noPoVal ? noPoVal.replace(/[^a-zA-Z0-9]/g, '') : ('ROW' + (i + 1)); currentId = 'DOC-' + cleanPo + '-' + (i + 1); sheet.getRange(i + 1, 1).setValue(currentId); row[0] = currentId; } var tanggalStr = formatTanggalCell(row[1]); var nominalVal = cleanNominalNumber(row[4]); var isChecked = parseBooleanVal(row[6]); var isLocked = parseBooleanVal(row[7]); result.push({ id: currentId, tanggal: tanggalStr, noPo: noPoVal, po: noPoVal, namaVendor: String(row[3] || ''), vendor: String(row[3] || ''), nominal: nominalVal, notes: String(row[5] || ''), isChecked: isChecked, isLocked: isLocked }); } } return result; } function saveBerkasBatch(items) { if (!items || items.length === 0) { return { status: 'error', message: 'Item berkas kosong' }; } var sheet = getBerkasSheet(); var rowsToAppend = []; for (var i = 0; i < items.length; i++) { var item = items[i]; var poVal = item.noPo || item.po || ''; var docId = item.id || ('DOC-' + (poVal ? poVal.replace(/[^a-zA-Z0-9]/g, '') : Date.now()) + '-' + i); rowsToAppend.push([ docId, item.tanggal || '', poVal, item.namaVendor || item.vendor || '', Number(item.nominal) || 0, item.notes || '', item.isChecked ? true : false, item.isLocked ? true : false ]); } if (rowsToAppend.length > 0) { var lastRow = sheet.getLastRow(); sheet.getRange(lastRow + 1, 1, rowsToAppend.length, 8).setValues(rowsToAppend); } return { status: 'success', addedCount: rowsToAppend.length }; } function updatePenerimaStatus(id, isCompleted, notes, optionalNoPo) { var sheet = getBerkasSheet(); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var rowId = String(data[i][0] || '').trim(); var rowPo = String(data[i][2] || '').trim(); if ((id && rowId === String(id).trim()) || (optionalNoPo && rowPo === String(optionalNoPo).trim())) { var rowNum = i + 1; if (!rowId && id) { sheet.getRange(rowNum, 1).setValue(id); } sheet.getRange(rowNum, 6).setValue(notes || ''); sheet.getRange(rowNum, 7).setValue(isCompleted ? true : false); if (isCompleted) { sheet.getRange(rowNum, 8).setValue(true); } return { status: 'success', message: 'Status berhasil diperbarui' }; } } return { status: 'error', message: 'Berkas tidak ditemukan: ID/PO ' + id }; } function updatePenerimaStatusBatch(updates) { if (!updates || updates.length === 0) { return { status: 'error', message: 'Daftar pembaruan kosong' }; } var sheet = getBerkasSheet(); var data = sheet.getDataRange().getValues(); var updatedCount = 0; var updateMap = {}; for (var u = 0; u < updates.length; u++) { var item = updates[u]; if (item.id) updateMap[String(item.id).trim()] = item; if (item.noPo) updateMap[String(item.noPo).trim()] = item; } for (var i = 1; i < data.length; i++) { var rowId = String(data[i][0] || '').trim(); var rowPo = String(data[i][2] || '').trim(); var matchedItem = updateMap[rowId] || updateMap[rowPo]; if (matchedItem) { var rowNum = i + 1; if (!rowId && matchedItem.id) { sheet.getRange(rowNum, 1).setValue(matchedItem.id); } sheet.getRange(rowNum, 6).setValue(matchedItem.notes || ''); sheet.getRange(rowNum, 7).setValue(matchedItem.isCompleted ? true : false); if (matchedItem.isCompleted) { sheet.getRange(rowNum, 8).setValue(true); } else if (matchedItem.isLocked !== undefined) { sheet.getRange(rowNum, 8).setValue(matchedItem.isLocked ? true : false); } updatedCount++; } } return { status: 'success', updatedCount: updatedCount }; } function updateDocumentDetails(id, updatedDoc) { if (!updatedDoc) return { status: 'error', message: 'Data update kosong' }; var sheet = getBerkasSheet(); var data = sheet.getDataRange().getValues(); var targetPo = updatedDoc.noPo || updatedDoc.po || ''; for (var i = 1; i < data.length; i++) { var rowId = String(data[i][0] || '').trim(); var rowPo = String(data[i][2] || '').trim(); if ((id && rowId === String(id).trim()) || (targetPo && rowPo === targetPo.trim())) { var rowNum = i + 1; var finalId = id || rowId || ('DOC-' + targetPo.replace(/[^a-zA-Z0-9]/g, '')); sheet.getRange(rowNum, 1).setValue(finalId); sheet.getRange(rowNum, 2).setValue(updatedDoc.tanggal || ''); sheet.getRange(rowNum, 3).setValue(targetPo); sheet.getRange(rowNum, 4).setValue(updatedDoc.namaVendor || updatedDoc.vendor || ''); sheet.getRange(rowNum, 5).setValue(Number(updatedDoc.nominal) || 0); sheet.getRange(rowNum, 6).setValue(updatedDoc.notes || ''); sheet.getRange(rowNum, 7).setValue(updatedDoc.isChecked ? true : false); sheet.getRange(rowNum, 8).setValue(updatedDoc.isLocked ? true : false); return { status: 'success', message: 'Detail berkas berhasil diperbarui' }; } } return { status: 'error', message: 'Berkas tidak ditemukan untuk diedit' }; } function unlockDocument(id, optionalNoPo) { var sheet = getBerkasSheet(); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var rowId = String(data[i][0] || '').trim(); var rowPo = String(data[i][2] || '').trim(); if ((id && rowId === String(id).trim()) || (optionalNoPo && rowPo === String(optionalNoPo).trim())) { sheet.getRange(i + 1, 8).setValue(false); return { status: 'success', message: 'Kunci berkas berhasil dibuka' }; } } return { status: 'error', message: 'Berkas tidak ditemukan' }; } function deleteDocument(id, optionalNoPo) { var sheet = getBerkasSheet(); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var rowId = String(data[i][0] || '').trim(); var rowPo = String(data[i][2] || '').trim(); if ((id && rowId === String(id).trim()) || (optionalNoPo && rowPo === String(optionalNoPo).trim())) { sheet.deleteRow(i + 1); return { status: 'success', message: 'Berkas berhasil dihapus' }; } } return { status: 'error', message: 'Berkas tidak ditemukan untuk dihapus' }; } function deleteGroupDate(ids, pos) { var sheet = getBerkasSheet(); var data = sheet.getDataRange().getValues(); var matchSet = {}; if (ids && ids.length) { for (var k = 0; k < ids.length; k++) matchSet[String(ids[k]).trim()] = true; } if (pos && pos.length) { for (var m = 0; m < pos.length; m++) matchSet[String(pos[m]).trim()] = true; } var deletedCount = 0; for (var i = data.length - 1; i >= 1; i--) { var rowId = String(data[i][0] || '').trim(); var rowPo = String(data[i][2] || '').trim(); if (matchSet[rowId] || matchSet[rowPo]) { sheet.deleteRow(i + 1); deletedCount++; } } return { status: 'success', deletedCount: deletedCount }; } function getLogsData() { var sheet = getSheetByName(SHEET_LOGBOOK); var data = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < data.length; i++) { var row = data[i]; if (row[0] || row[1] || row[3]) { result.push({ id: String(row[0] || ('LOG-' + i)), timestamp: formatTanggalCell(row[1]), role: String(row[2] || 'system'), action: String(row[3] || ''), details: String(row[4] || '') }); } } return result.reverse(); } function addLogbookEntry(logData) { if (!logData) return { status: 'error', message: 'Log data kosong' }; var sheet = getSheetByName(SHEET_LOGBOOK); sheet.appendRow([ logData.id || ('LOG-' + Date.now()), logData.timestamp || new Date().toLocaleString('id-ID'), logData.role || 'system', logData.action || 'Aktivitas', logData.details || '-' ]); return { status: 'success' }; } function clearLogbook(idsToDelete) { var sheet = getSheetByName(SHEET_LOGBOOK); if (idsToDelete && idsToDelete.length > 0) { var data = sheet.getDataRange().getValues(); var idSet = {}; for (var k = 0; k < idsToDelete.length; k++) idSet[String(idsToDelete[k])] = true; for (var i = data.length - 1; i >= 1; i--) { if (idSet[String(data[i][0])]) { sheet.deleteRow(i + 1); } } } else { var lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.deleteRows(2, lastRow - 1); } } return { status: 'success' }; } function getBerkasSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(PRIMARY_SHEET_NAME); if (!sheet) { sheet = ss.getSheetByName(ALTERNATE_SHEET_NAME); } if (!sheet) { setupSheetsIfNotExist(); sheet = ss.getSheetByName(PRIMARY_SHEET_NAME); } return sheet; } function getSheetByName(name) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(name); if (!sheet) { setupSheetsIfNotExist(); sheet = ss.getSheetByName(name); } return sheet; } function setupSheetsIfNotExist() { var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) { throw new Error("Spreadsheet aktif tidak ditemukan. Pastikan script ini terikat (bound) pada Google Sheets."); } var berkasSheet = ss.getSheetByName(PRIMARY_SHEET_NAME) || ss.getSheetByName(ALTERNATE_SHEET_NAME); if (!berkasSheet) { berkasSheet = ss.insertSheet(PRIMARY_SHEET_NAME); berkasSheet.appendRow(["ID", "Tanggal", "No PO", "Nama Vendor", "Nominal", "Catatan Penerimaan", "Ceklis", "Status Terkunci"]); berkasSheet.getRange("1:1").setFontWeight("bold").setBackground("#1e293b").setFontColor("#ffffff"); } var logSheet = ss.getSheetByName(SHEET_LOGBOOK); if (!logSheet) { logSheet = ss.insertSheet(SHEET_LOGBOOK); logSheet.appendRow(["ID", "Timestamp", "Role", "Action", "Details"]); logSheet.getRange("1:1").setFontWeight("bold").setBackground("#1e293b").setFontColor("#ffffff"); } } function formatTanggalCell(val) { if (!val) return ''; if (val instanceof Date) { var d = val.getDate(); var m = val.getMonth() + 1; var y = val.getFullYear(); return (d < 10 ? '0' + d : d) + '/' + (m < 10 ? '0' + m : m) + '/' + y; } return String(val).trim(); } function cleanNominalNumber(val) { if (typeof val === 'number') return val; if (!val) return 0; var cleaned = String(val).replace(/[^0-9.-]+/g, ""); return Number(cleaned) || 0; } function parseBooleanVal(val) { if (val === true || val === 1) return true; var str = String(val || '').trim().toUpperCase(); return str === 'TRUE' || str === '1' || str === 'YA' || str === 'CHECKED'; } function createJsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
